async function handleSync() {
  if (!fbAuth || !fbProvider) {
    openFbKeyModal();
    return;
  }
  if (!fbUser) {
    try {
      setSyncStatus('info', 'Iniciando sesión con Google…');

  } else {
    await doSync();
  }

async function doSync() {
  if (!fbUser || !fbDb) return;
  setSyncStatus('info', 'Sincronizando historial…');
  get('btn-sync').classList.add('syncing');

  try {
    const docRef = fbDb.collection('editor').doc(fbUser.uid);
    const snap   = await docRef.get();
    const local  = [...versions];
    const localSettings = JSON.parse(JSON.stringify(editorSettings));
    const localSettingsTs = parseInt(localStorage.getItem('castroverde_editor_settings_ts') || '0');

    if (snap.exists) {
      const data   = snap.data();
      const remote = data.versions || [];
      const merged = mergeVersions(local, remote);
      versions = merged;
      saveVersions();

      var mergedSettings = localSettings;
      if (data.editorSettings && data.editorSettingsTs) {
        if (data.editorSettingsTs > localSettingsTs) {
          mergedSettings = data.editorSettings;
          editorSettings = mergedSettings;
          saveEditorSettings(editorSettings);
          localStorage.setItem('castroverde_editor_settings_ts', data.editorSettingsTs);
        }
      }

      await docRef.set({
        versions: merged,
        editorSettings: mergedSettings,
        editorSettingsTs: Math.max(localSettingsTs, data.editorSettingsTs || 0),
        updatedAt: firebase.firestore.FieldValue.serverTimestamp()
      });
      setSyncStatus('ok', '✓ Sincronizado — ' + merged.length + ' prompt' + (merged.length !== 1 ? 's' : '') + ' en historial');
    } else {
      await docRef.set({
        versions: local,
        editorSettings: localSettings,
        editorSettingsTs: localSettingsTs || Date.now(),
        updatedAt: firebase.firestore.FieldValue.serverTimestamp()
      });
      setSyncStatus('ok', '✓ Historial subido por primera vez — ' + local.length + ' prompt' + (local.length !== 1 ? 's' : ''));
    }

    renderHistoryList();

  } catch(err) {
    console.error('Sync error:', err);
    setSyncStatus('err', 'Error al sincronizar: ' + err.message);
  }

  get('btn-sync').classList.remove('syncing');
}

function mergeVersions(local, remote) {
  const map = new Map();
  for (const v of remote) if (v.ts) map.set(v.ts, v);
  for (const v of local)  if (v.ts) map.set(v.ts, v);
  return Array.from(map.values()).sort((a, b) => a.ts - b.ts);
}

function setSyncStatus(type, msg) {
  const el = get('sync-status');
  get('sync-status-msg').textContent = msg;
  el.className = 'visible ' + type;
  if (type === 'ok') setTimeout(closeSyncStatus, 4000);
}

function closeSyncStatus() {
  get('sync-status').className = '';
  updateSyncUI();
}

function buildSystem(){
  var parte = get('parte-libro') ? get('parte-libro').value : 'primera';
  var pl = parte==='segunda' ? 'SEGUNDA PARTE (tematica y formal)' : 'PRIMERA PARTE (narrativa divulgativa)';
  var gen = getInstrucGen().trim();
  var extra = '\n\nPARTE DEL LIBRO: Este fragmento pertenece a la '+pl+' del libro.';
  if(gen) extra += '\n\nINSTRUCCIONES GENERALES DEL AUTOR:\n'+gen;
  return editorSettings.system+extra;
}

function buildMsg(text,notes,tipo,frag){
  var lbl={completo:'Revision completa',estilo:'Estilo y voz',ortografia:'Ortografia y gramatica',historico:'Coherencia historica',claridad:'Claridad y estructura'};
  var tpl = editorSettings.promptAnalisis || DEFAULT_PROMPT_ANALISIS;
  var instrucFrag = (frag&&frag.trim()) ? 'INSTRUCCIONES ESPECIFICAS:\n'+frag+'\n\n' : '';
  var notasPart = notes ? 'NOTAS AL PIE (contexto, no modificar):\n\'\'\'\n'+notes+'\n\'\'\'\n\n' : '';
  return tpl
    .replace(/\{\{tipo\}\}/g, lbl[tipo]||'Revision completa')
    .replace(/\{\{instruc_frag\}\}/g, instrucFrag)
    .replace(/\{\{notas\}\}/g, notasPart)
    .replace(/\{\{texto\}\}/g, text);
}

// Nueva función: mostrar el prompt en lugar de llamar a la API
function showPrompt(msg, systemPrompt, onSucces, onError) {
  // Simulamos una respuesta exitosa con el prompt como texto
  var fullPrompt = "=== SYSTEM PROMPT ===\n" + systemPrompt + "\n\n=== USER MESSAGE ===\n" + msg;

function analyzeText(){
  var text=get('input-text').value.trim();
  var notes=get('notes-text').value.trim();
  if(!text){ showError('Introduce un texto para generar el prompt.'); return; }
  if(text.length<50){ showError('El fragmento es demasiado corto.'); return; }
  var instrucFrag = get('instruc-fragmento') ? get('instruc-fragmento').value.trim() : '';
  var msg = buildMsg(text, notes, get('analysis-type').value, instrucFrag);
  var system = buildSystem();
  showLoading('Generando prompt…'); hideError();
  // Llamada simulada: mostramos el prompt
  showPrompt(msg, system, function(response){
    hideLoading();
    currentResult = {
      improved: response,
      report: {
        debug_info: "Este es el prompt que se enviaría a Claude. No se ha llamado a ninguna API externa.",
        system_prompt: system,
        user_message: msg
      },
      original: text,
      name: get('fragment-name').value.trim() || ('Prompt · ' + dateStr(new Date())),
      isPrompt: true
    };
    renderResult(currentResult);
    autoSaveVersion();
  }, function(err){ hideLoading(); showError(err); });
}

function requestVerdict(){
  var text=get('input-text').value.trim();
  var notes=get('notes-text').value.trim();
  if(!text){ showError('Introduce el fragmento para el veredicto.'); return; }
  var tpl = editorSettings.promptVeredicto || DEFAULT_PROMPT_VEREDICTO;
  var notasPart = notes ? 'NOTAS AL PIE:\n"""\n'+notes+'\n"""\n\n' : '';
  var msg = tpl
    .replace(/\{\{notas\}\}/g, notasPart)
    .replace(/\{\{texto\}\}/g, text);
  var system = buildSystem();
  showLoading('Generando prompt de veredicto…'); hideError();
  showPrompt(msg, system, function(response){
    hideLoading();
    currentResult = {
      improved: response,
      report: {
        debug_info: "Prompt para solicitar veredicto editorial. No se ha llamado a la API.",
        system_prompt: system,
        user_message: msg
      },
      original: text,
      name: get('fragment-name').value.trim() || ('Veredicto prompt · ' + dateStr(new Date())),
      isPrompt: true
    };
    renderResult(currentResult);
    autoSaveVersion();
  }, function(err){ hideLoading(); showError(err); });
}

function renderHistoryList(){
  var list=get('history-list');
  if(!versions.length){ list.innerHTML='<div class="history-empty">Sin prompts guardados.<br>Genera y guarda<br>prompts para verlos aquí.</div>'; return; }
  var html='';
  for(var i=versions.length-1;i>=0;i--){
    var v=versions[i];
    var instrucPreview = v.instrucFragmento ? '<div style="font-family:var(--font-ui);font-size:8px;color:var(--accent);margin-top:3px;overflow:hidden;text-overflow:ellipsis;white-space:nowrap;">✦ '+esc(v.instrucFragmento.substring(0,60))+(v.instrucFragmento.length>60?'…':'')+'</div>' : '';
    html+='<div class="version-item"><div class="version-meta"><span class="version-name">'+esc(v.name)+'</span><span class="version-date">'+v.date+'</span></div><div class="version-preview">'+esc(v.original.substring(0,80))+'…</div>'+instrucPreview+'<div class="version-actions"><button class="btn btn-ghost" style="font-size:9px;padding:3px 9px;" onclick="loadVersion('+i+')">Cargar texto</button><button class="btn btn-danger" style="font-size:9px;padding:3px 9px;" onclick="deleteVersion('+i+')">Borrar</button></div></div>';

function autoSaveVersion(){
  if(!currentResult) return;
  var instrucFrag = get('instruc-fragmento') ? get('instruc-fragmento').value : '';
  versions.push({name:currentResult.name,original:currentResult.original,improved:currentResult.improved,report:currentResult.report,instrucFragmento:instrucFrag,date:dateStr(new Date()),ts:Date.now()});
  saveVersions(); renderHistoryList();
}

function loadVersion(idx){
  var v=versions[idx]; if(!v) return;
  get('input-text').value=v.original; get('fragment-name').value=v.name; updateCharCount();
  if(get('instruc-fragmento')) get('instruc-fragmento').value = v.instrucFragmento || '';
  currentResult={improved:v.improved,report:v.report,original:v.original,name:v.name, isPrompt:true};
  renderResult(currentResult); closeHistory();
}

function deleteVersion(idx){ versions.splice(idx,1); saveVersions(); renderHistoryList(); }

get('file-input').addEventListener('change', function(e){
  var file=e.target.files[0]; if(!file) return;
  var name=file.name, ext=name.split('.').pop().toLowerCase();
  if(ext==='txt'){
    var reader=new FileReader();
    reader.onload=function(ev){ get('input-text').value=ev.target.result.substring(0,4000); updateCharCount(); if(!get('fragment-name').value) get('fragment-name').value=name.replace('.txt',''); };
    reader.readAsText(file,'UTF-8');
  } else if(ext==='docx'){
    var reader2=new FileReader();
    reader2.onload=function(ev){
      if(typeof mammoth==='undefined'){ showError('La librería de Word no está disponible.'); return; }
      mammoth.extractRawText({arrayBuffer: ev.target.result}).then(function(result){
        get('input-text').value=result.value.substring(0,4000); updateCharCount();
        if(!get('fragment-name').value) get('fragment-name').value=name.replace('.docx','');
      }).catch(function(){ showError('No se pudo leer el archivo .docx.'); });
    };
    reader2.readAsArrayBuffer(file);
  } else { showError('Formato no soportado. Usa archivos .txt o .docx'); }
  e.target.value='';
});

    const localSettingsTs = parseInt(localStorage.getItem('castroverde_editor_settings_ts') || '0');

    if (snap.exists) {
      const data   = snap.data();
      const remote = data.versions || [];
      const merged = mergeVersions(local, remote);
      versions = merged;

      var mergedSettings = localSettings;

          editorSettings = mergedSettings;
          saveEditorSettings(editorSettings);
          localStorage.setItem('castroverde_editor_settings_ts', data.editorSettingsTs);
        }
      }

      await docRef.set({
        versions: merged,
        editorSettings: mergedSettings,
        editorSettingsTs: Math.max(localSettingsTs, data.editorSettingsTs || 0),
        updatedAt: firebase.firestore.FieldValue.serverTimestamp()
      });
      setSyncStatus('ok', '✓ Sincronizado — ' + merged.length + ' prompt' + (merged.length !== 1 ? 's' : '') + ' en historial');

function mergeVersions(local, remote) {

  return Array.from(map.values()).sort((a, b) => a.ts - b.ts);
}

  function buildSceneDefinitionBlock(sd) {
    const requiredFields = [ sd.accion.trim(), sd.personajes.trim(), sd.contexto.trim(),
                             sd.entorno.trim(), sd.foco.trim(), sd.densidad.trim() ];
    const hasAny = requiredFields.some(f => f);
    if (!hasAny) return '';

    const densityLabels = { intimate: 'intimate (2–4 figures)', medium: 'medium (5–10 figures)', crowd: 'crowd' };
    const densityText = densityLabels[sd.densidad] || sd.densidad;

    let block = 'SCENE DEFINITION — REQUIRED:\n\nThis image must depict a specific historical moment.\n\n';
    if (sd.accion.trim())     block += `* Action: ${sd.accion.trim()}\n`;
    if (sd.personajes.trim()) block += `* Characters: ${sd.personajes.trim()}\n`;
    if (sd.contexto.trim())   block += `* Social context: ${sd.contexto.trim()}\n`;
    if (sd.entorno.trim())    block += `* Environment: ${sd.entorno.trim()}\n`;
    if (sd.foco.trim())       block += `* Central focus: ${sd.foco.trim()}\n`;
    if (densityText)          block += `* Density: ${densityText}\n`;

    const opts = [];
    if (sd.composicion) opts.push(`* Composition: ${sd.composicion}`);
    if (sd.actitud)     opts.push(`* Character attitude: ${sd.actitud}`);
    if (sd.peso)        opts.push(`* Environment weight: ${sd.peso}`);
    if (sd.prioridad)   opts.push(`* Priority: ${sd.prioridad}`);
    if (opts.length) block += '\n' + opts.join('\n') + '\n';

    if (sd.negativos.trim()) block += `\n* Avoid: ${sd.negativos.trim()}\n`;

    block += '\nThe scene must be visually clear, historically plausible, and free of generic or ambiguous representation.';
    return block;
  }
